1.2 信号流

操作路径AXI4 通道
取指fetch → arbiter → SlaveAR(addr=PC) → R(data=instr)
loadmem → arbiter → SlaveAR(addr=ALU) → R(data=rdata)
storemem → arbiter → SlaveAW(addr=ALU) + W(data=rs2) → B(resp)

2.6 AR 通道结构体

📄 axi4.sv · 第 38-45 行
typedef struct packed {
    logic [`AXI4_ID_W-1:0]    id;      // 事务 ID
    logic [`AXI4_ADDR_W-1:0]  addr;    // 读地址
    logic [`AXI4_LEN_W-1:0]   len;     // burst 长度 - 1
    logic [`AXI4_SIZE_W-1:0]  size;    // 每拍字节数 = 2^size

字段NPC 设定值说明
id4'h0固定 0，无需区分事务
addrPC 或 ALU 结果取指用 PC，访存用 ALU
len8'h0单次传输（1 拍）
size3'b0104 字节
burst2'b01 (INCR)递增地址

4.1 模块声明

📄 fetch.sv · 第 4-36 行
module fetch #(
    parameter AW = 32,
    parameter DW = 32
)(
    input  logic          clk,
    input  logic          rst_n,
    input  logic [AW-1:0] pc_pointer,

    // AXI4 AR 通道
    output logic          ifu_arvalid,
    input  logic          ifu_arready,
    output logic [AW-1:0] ifu_araddr,
    output logic [2:0]    ifu_arsize,
    output logic [7:0]    ifu_arlen,

    // AXI4 R 通道
    input  logic          ifu_rvalid,
    output logic          ifu_rready,
    input  logic [DW-1:0] ifu_rdata,
    input  logic [1:0]    ifu_rresp,
    input  logic          ifu_rlast,

端口分四组：时钟复位、AXI4 AR（读地址）、AXI4 R（读数据）、流水线控制。AR 和 R 是独立的 AXI4 通道，各自有自己的 valid/ready 握手。

4.2 状态机定义

📄 fetch.sv · 第 38-46 行
typedef enum logic [1:0] {
    S_REQ       = 2'b00,  // 发出 arvalid，等 arready
    S_WAIT_RESP = 2'b01,  // 已握手，等 rvalid
    S_DONE      = 2'b10   // 指令就绪，等流水线消费
} state_t;

state_t state, state_next;
logic [DW-1:0] instr_latch;
logic [1:0]    rresp_latch;

3 个状态对应 AXI4 读事务的 3 个阶段。instr_latch 锁存指令，rresp_latch 锁存响应码。

4.3 状态转移

📄 fetch.sv · 第 48-78 行
// 时序逻辑：状态寄存器
always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) state <= S_REQ;
    else state <= state_next;
end

// 组合逻辑：下一状态
always_comb begin
    state_next = state;
    case (state)
        S_REQ: begin
            if (flush)            state_next = S_REQ;       // flush 回到空闲
            else if (ifu_arvalid && ifu_arready) state_next = S_WAIT_RESP; // AR 握手成功
        end
        S_WAIT_RESP: begin
            if (flush)            state_next = S_REQ;
            else if (ifu_rvalid && ifu_rready) state_next = S_DONE;    // R 握手成功
        end
        S_DONE: begin
            if (flush)            state_next = S_REQ;
            else if (!stall)      state_next = S_REQ;       // 流水线消费后回空闲
        end
        default: state_next = S_REQ;
    endcase
end

任何状态都可以被 flush（分支跳转）打断，回到 S_REQ 重新取指。stall（load-use 冲突）阻止 S_DONE → S_REQ 的转移。

4.4 响应锁存

📄 fetch.sv · 第 80-92 行
always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        instr_latch <= `INSTR_NOP;
        rresp_latch <= 2'b00;
    end else if (state == S_WAIT_RESP && ifu_rvalid && ifu_rready && !flush) begin
        instr_latch <= ifu_rdata;    // 锁存指令
        rresp_latch <= ifu_rresp;    // 锁存响应码
    end else if (flush) begin
        instr_latch <= `INSTR_NOP;
        rresp_latch <= 2'b00;       // flush 时清除
    end
end

在 R 通道握手成功的同一周期锁存 ifu_rdata 和 ifu_rresp。flush 时两者都清除为 NOP/OKAY。

4.5 输出赋值

📄 fetch.sv · 第 94-108 行

6axi_arbiter.sv — AXI4 合并器

6.1 设计原则

部分策略原因
AR（读地址）组合 MUX，LSU 优先流水线中 IFU 被 lsu_busy 阻塞，不会同时发请求
R（读数据）sel_lsu_ar 路由1-bit 寄存器记录谁发了 AR
AW/W/B（写）直通仅 LSU 使用，无需仲裁

6.2 端口声明

📄 axi_arbiter.sv · 第 12-101 行
module axi_arbiter (
    input  logic clk, rst_n,

    // IFU AXI4 端口 (只读)
    input  logic        ifu_arvalid,
    output logic        ifu_arready,
    input  logic [31:0] ifu_araddr,
    input  logic [2:0]  ifu_arsize, ifu_arlen [7:0], ifu_arburst, ifu_arid [3:0],
    output logic        ifu_rvalid,
    input  logic        ifu_rready,
    output logic [31:0] ifu_rdata,
    output logic [1:0]  ifu_rresp,
    output logic        ifu_rlast,

    // LSU AXI4 端口 (读写)
    // AR + R + AW + W + B（共 26 个信号）

    // AXI4 Master 端口 (连接 SoC)
    // AR + R + AW + W + B（共 20 个信号）
);

三组端口：IFU（AR/R）、LSU（AR/R + AW/W/B）、Master（AR/R + AW/W/B）。IFU 只有读通道，LSU 有读写通道。

6.3 AR 通道选择寄存器

📄 axi_arbiter.sv · 第 110-121 行
logic sel_lsu_ar;  // 1 = LSU 获得 AR 通道

always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n)
        sel_lsu_ar <= 1'b0;
    else if (m_arvalid && m_arready) begin
        sel_lsu_ar <= 1'b0;  // 握手后清除（下次默认 IFU）
    end else if (lsu_arvalid && !ifu_arvalid) begin
        sel_lsu_ar <= 1'b1;  // LSU 优先
    end
end

这是 arbiter 中唯一的时序逻辑。记录"当前 R 通道的响应应该路由给谁"。握手后自动清除，确保下次默认路由给 IFU。

6.4 AR 通道 MUX

📄 axi_arbiter.sv · 第 123-133 行
// AR MUX
assign m_arvalid = lsu_arvalid | ifu_arvalid;
assign m_araddr  = lsu_arvalid ? lsu_araddr  : ifu_araddr;
assign m_arid    = lsu_arvalid ? lsu_arid    : ifu_arid;
assign m_arlen   = lsu_arvalid ? lsu_arlen   : ifu_arlen;
assign m_arsize  = lsu_arvalid ? lsu_arsize  : ifu_arsize;
assign m_arburst = lsu_arvalid ? lsu_arburst : ifu_arburst;

// AR ready 路由

assign ifu_arready = m_arready && !lsu_arvalid;

LSU 有请求时优先选择 LSU，否则选择 IFU。arready 路由确保只有被选中的 master 收到 ready。

6.5 R 通道路由
